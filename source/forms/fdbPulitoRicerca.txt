Private Sub cmdCerca_Click()
On Error GoTo Err_cmdCerca_Click

    Dim stDocName As String
    Dim stLinkCriteria As String
    Dim swhere As String
    

    If Not cmdCategorie = "" Then
        If Not cmdPaesaggio = "" Then
            swhere = "[Ambito]='" & cmdCategorie & "' AND Paesaggio='" & cmdPaesaggio & "'"
        Else
            swhere = "[Ambito]='" & cmdCategorie & "'"
        End If
       
    Else
        If Not cmdPaesaggio = "" Then
            swhere = "Paesaggio='" & cmdPaesaggio & "'"
        Else
            swhere = "Paesaggio like '*'"
        End If
    End If
    
    
    stDocName = "fdbPulito"
    DoCmd.OpenForm stDocName, , , swhere
